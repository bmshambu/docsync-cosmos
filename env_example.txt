# ── LLM provider ───────────────────────────────────────────────
# Set to "google" (default) or "azure_openai"
LLM_PROVIDER=google

# ── Google AI ──────────────────────────────────────────────────
# Get a free key from https://aistudio.google.com/app/apikey
GOOGLE_API_KEY=""


# ── Models (Gemini) ────────────────────────────────────────────
# Gemini 2.5 Flash is the default for all three steps.
MODEL_EXTRACT="gemini-2.5-flash"
MODEL_SUMMARY="gemini-2.5-flash"
MODEL_QUERY="gemini-2.5-flash"

# ── Pipeline tuning ────────────────────────────────────────────
# How many documents to send to the LLM in parallel during entity extraction.
MAX_LLM_CONCURRENCY=5
CHUNK_SIZE=400
CHUNK_OVERLAP=50
# Increase for reasoning models (o1, GPT-4.5, gpt-5.5) that use thinking tokens
MAX_EXTRACT_TOKENS=16000
MAX_QUERY_TOKENS=4096
# Thinking models (Gemini 2.5, GPT reasoning) burn tokens on reasoning before
# producing output — too low a cap silently produces EMPTY summaries
MAX_SUMMARY_TOKENS=8192

# ── Retrieval caps (see README "How retrieve() works") ─────────
# Precision caps: what gets RETRIEVED per question
TOP_ENTITIES=10            # matched entities kept after ranking
TOP_COMMUNITIES=3          # community summaries used for global queries
TOP_CHUNKS=4               # source chunks cited per answer
CHUNK_CANDIDATE_LIMIT=200  # backend candidate cap before ranking
# Prompt caps: what reaches the LLM out of the retrieved set
MAX_PROMPT_ENTITIES=8
MAX_PROMPT_RELATIONSHIPS=15
# Filename-as-question matching (response-library docs are named as the
# question they answer — matched docs' chunks are guaranteed into the context)
TITLE_MATCH_DOCS=3
TITLE_MATCH_THRESHOLD=0.4

# AZURE_OPENAI_API_KEY=your-azure-key-here
# AZURE_OPENAI_ENDPOINT=https://your-resource.openai.azure.com/
# AZURE_OPENAI_DEPLOYMENT=gpt-4o
# AZURE_OPENAI_API_VERSION=2024-02-15-preview
# Reasoning models (GPT-5.x, o1/o3): keep this LOW/MINIMAL for extraction &
# summaries — those are mechanical tasks, and high reasoning wastes the token
# budget (empty output → parse failures), time (~2 min/doc), and cost.
# Leave EMPTY for non-reasoning deployments (gpt-4o) so it isn't sent.
# AZURE_REASONING_EFFORT=low

# ── Storage ────────────────────────────────────────────────────
# Where the app writes extracted_text/, chunks/, graph/ etc.
DATA_DIR=data

# ── Graph persistence backend ──────────────────────────────────
# "file"   → JSON/md files under DATA_DIR (today's behaviour)
# "cosmos" → Azure Cosmos DB NoSQL
STORAGE_BACKEND=cosmos

# ── Azure Cosmos DB (used when STORAGE_BACKEND=cosmos) ─────────
# PoC account (personal subscription — when moving to the office account,
# change ONLY these three values; nothing else in the app changes):
#   Portal → graph-rag-docsync-cosmos → Keys → URI + Primary Key
COSMOS_ENDPOINT=https://xxxxx.documents.azure.com:443/
COSMOS_KEY=""
COSMOS_DATABASE=graphrag

# ── Azure Blob Storage (document source) ───────────────────────────────────────
# Set both vars to switch from local folder to your blob container.
# Get the connection string from: Azure Portal → Storage Account → Access Keys
AZURE_STORAGE_CONNECTION_STRING=""
AZURE_STORAGE_CONTAINER_NAME=rfp-docs


# database with SHARED throughput (this is what the portal wouldn't offer)
az cosmosdb sql database create \
  --account-name graph-rag-docsync-cosmos -g graph_rag_poc \
  -n graphrag --throughput 1000

# five containers — no --throughput flag, so they inherit the shared 1000 RU/s
az cosmosdb sql container create --account-name graph-rag-docsync-cosmos -g graph_rag_poc \
  -d graphrag -n entities      --partition-key-path /type
az cosmosdb sql container create --account-name graph-rag-docsync-cosmos -g graph_rag_poc \
  -d graphrag -n relationships --partition-key-path /source_doc
az cosmosdb sql container create --account-name graph-rag-docsync-cosmos -g graph_rag_poc \
  -d graphrag -n chunks        --partition-key-path /doc_id
az cosmosdb sql container create --account-name graph-rag-docsync-cosmos -g graph_rag_poc \
  -d graphrag -n communities   --partition-key-path /graph_id
az cosmosdb sql container create --account-name graph-rag-docsync-cosmos -g graph_rag_poc \
  -d graphrag -n jobs          --partition-key-path /id
